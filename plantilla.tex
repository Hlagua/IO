	\subsection{Objetivos del proyecto}
	
	Al completar este proyecto, el estudiante será capaz de:
	
	\begin{enumerate}
		\item Formular un problema de transporte como un modelo de Programación Lineal.
		\item Implementar la solución en R.
		\item Calcular tiempos tempranos, tardíos, holguras y ruta crítica con PERT/CPM.

	\subsection{Contexto del problema}
	
	Una planta embotelladora de agua purificada tiene \textbf{3 centros de producción} y debe abastecer a \textbf{4 centros de distribución (CD)}. Adicionalmente, debe ejecutar un proyecto de instalación de una nueva red de distribución.

	\section{Parte 1: Modelo de transporte}
	
	\subsection{Descripción del problema}
	
	Se debe determinar cuántos miles de litros enviar desde cada planta a cada CD para minimizar el costo total de transporte, respetando las capacidades de las plantas y las demandas de los CD.
	
	\subsection{Datos proporcionados}
	
	\subsubsection{Capacidad de producción (miles de litros/semana)}
	
	\begin{table}[H]
		\centering
		\begin{tabular}{|c|c|}
			\hline
			\textbf{Planta} & \textbf{Capacidad} \\
			\hline
			A & 50 \\
			B & 60 \\
			C & 40 \\
			\hline
		\end{tabular}
		\hspace{2cm}
		\begin{tabular}{|c|c|}
			\hline
			\textbf{CD} & \textbf{Demanda} \\
			\hline
			CD1 & 30 \\
			CD2 & 45 \\
			CD3 & 35 \\
			CD4 & 40 \\
			\hline
		\end{tabular}
	\end{table}
	
	\subsubsection{Costos de transporte (\$/miles de litros)}
	
	\begin{table}[H]
		\centering
		\begin{tabular}{|c|cccc|}
			\hline
			\textbf{Origen $\backslash$ Destino} & CD1 & CD2 & CD3 & CD4 \\
			\hline
			Planta A & 8 & 6 & 10 & 9 \\
			Planta B & 9 & 12 & 13 & 7 \\
			Planta C & 14 & 9 & 16 & 5 \\
			\hline
		\end{tabular}
	\end{table}
	
	\subsection{Tareas específicas}
	
	\subsubsection{Formulación matemática}
	
	Escriba el modelo completo con:
	
	\begin{itemize}
		\item Variables de decisión (definición clara)
		\item Función objetivo
		\item Restricciones de oferta %(3 restricciones)
		\item Restricciones de demanda %(4 restricciones)
		\item Restricción de no negatividad
	\end{itemize}
	
	\noindent \textbf{Formato sugerido:}
	
	\begin{align}
		\text{Variables: } & x_{ij} = \text{miles de litros enviados desde planta } i \text{ a CD } j \\
		& i \in \{A, B, C\}, \quad j \in \{1, 2, 3, 4\} \\
		\text{Min } Z = & \sum_{i=1}^{3}\sum_{j=1}^{4} c_{ij}x_{ij}
	\end{align}
	
	\subsubsection{Implementación en R}
	
	Cree un script que:
	
	\begin{enumerate}
		\item Cargue el paquete \texttt{lpSolve}
		\item Defina la matriz de costos
		\item Defina los vectores de capacidad y demanda
		\item Ejecute \texttt{lp.transport()} con los parámetros correctos
		\item Almacene el resultado en una variable
	\end{enumerate}
	
	\noindent \textbf{Estructura base:}
	
	\begin{lstlisting}[language=R, caption=Estructura para el modelo de transporte]
		library(lpSolve)
		
		# Matriz de costos (3 filas, 4 columnas)
		costos <- matrix(c(
		8, 6, 10, 9,
		9, 12, 13, 7,
		14, 9, 16, 5
		), nrow = 3, byrow = TRUE)
		
		# Capacidades (oferta)
		capacidad <- c(50, 60, 40)
		
		# Demandas
		demanda <- c(30, 45, 35, 40)
		
		# Resolver modelo de transporte
		transporte <- lp.transport(
		cost.mat = costos,
		direction = "min",
		row.signs = rep("<=", 3),
		row.rhs = capacidad,
		col.signs = rep(">=", 4),
		col.rhs = demanda
		)
	\end{lstlisting}
	
	\subsubsection{Extracción de resultados}
	
	Extraiga y reporte:
	
	\begin{itemize}
		\item Costo total mínimo por semana
		\item Matriz de envíos óptimos
		\item Verificación de que cada planta no excede su capacidad
		\item Verificación de que cada CD recibe al menos su demanda
	\end{itemize}
	
	\subsubsection{Interpretación}
	
	Responda:
	
	\begin{enumerate}
		\item ¿Qué planta envía más producto? ¿Por qué?
		\item ¿Qué ruta es la más utilizada? ¿Qué ruta no se utiliza?
		\item ¿Por qué la Planta C solo envía a un CD?
		\item ¿Qué pasaría si la demanda de CD4 aumenta a 50?
	\end{enumerate}

	\section{Parte 2: Gestión del proyecto (PERT/CPM)}
	
	\subsection{Descripción del problema}
	
	La empresa debe ejecutar un proyecto de instalación de una nueva red de distribución con 7 actividades. El proyecto debe durar máximo \textbf{40 días hábiles} (8 semanas).
	
	\subsection{Datos proporcionados}
	
	\begin{table}[H]
		\centering
		\begin{tabular}{|c|l|c|l|}
			\hline
			\textbf{Actividad} & \textbf{Descripción} & \textbf{Duración (días)} & \textbf{Predecesor(es)} \\
			\hline
			A & Diseño de ruta & 5 & -- \\
			B & Compra de válvulas & 3 & A \\
			C & Instalación en CD1 & 4 & B \\
			D & Instalación en CD2 & 6 & B \\
			E & Pruebas de flujo & 2 & C, D \\
			F & Capacitación & 3 & E \\
			G & Inicio de operación & 1 & F \\
			\hline
		\end{tabular}
	\end{table}
	
	\subsection{Tareas específicas}
	
	\subsubsection{Construcción de la red}
	
	Dibuje (a mano o con R) el diagrama de red mostrando:
	
	\begin{itemize}
		\item Nodos (eventos/inicios de actividades)
		\item Flechas (actividades con su duración)
		\item Relaciones de precedencia claras
	\end{itemize}
	
	\subsubsection{Cálculo de tiempos tempranos (Forward Pass)}
	
	Calcule para cada actividad:
	
	\begin{itemize}
		\item \textbf{Tiempo de inicio más temprano (TE)}
		\item \textbf{Tiempo de finalización más temprano (TF)}
	\end{itemize}
	
%	\noindent \textbf{Procedimiento:}

%	\begin{enumerate}
%		\item Actividad A: TE = 0
%		\item Actividad B: TE = TF de su predecesor
%		\item Actividad C y D: TE = TF de B
%		\item Actividad E: TE = max(TF de C, TF de D)
%		\item Continuar hasta G
%	\end{enumerate}
	
	\subsubsection{Cálculo de tiempos tardíos (Backward Pass)}
	
	Calcule para cada actividad:
	
	\begin{itemize}
		\item \textbf{Tiempo de inicio más tardío (TL)} (sin retrasar el proyecto)
		\item \textbf{Tiempo de finalización más tardío (TFL)}
	\end{itemize}
	
%	\noindent \textbf{Procedimiento:}
%	
%	\begin{enumerate}
%		\item Duración del proyecto = TF de G
%		\item Actividad G: TL = duración\_proyecto - duración
%		\item Actividad F: TL = TL de G - duración de F
%		\item Actividad E: TL = TL de F - duración de E
%		\item Actividad D y C: TL = TL de E - duración respectiva
%		\item Actividad B: TL = min(TL de C, TL de D) - duración de B
%		\item Actividad A: TL = TL de B - duración de A
%	\end{enumerate}
%	
	\subsubsection{Cálculo de holguras y ruta crítica}
	
	Para cada actividad, calcule:
	
	\begin{itemize}
		\item \textbf{Holgura total} = TL - TE (o TFL - TF)
		\item \textbf{¿Es crítica?} (holgura = 0)
	\end{itemize}
	
	Identifique la \textbf{ruta crítica} (secuencia de actividades con holgura 0).
	
	\subsubsection{Diagrama de Gantt}
	
	Construya un diagrama de Gantt mostrando:
	
	\begin{itemize}
		\item Eje X: Días (0 a duración total del proyecto)
		\item Eje Y: Actividades (A a G)
		\item Barras horizontales: duración de cada actividad
		\item Diferentes colores para actividades críticas y no críticas
		\item Holguras visibles (espacios vacíos antes de iniciar)
	\end{itemize}
	
	\subsubsection{Verificación del plazo}
	
	Responda:
	
	\begin{enumerate}
		\item ¿Cuál es la duración total del proyecto en días?
		\item ¿Cumple con el límite de 40 días? Si no, ¿cuántos días excede?
		\item Si excede, proponga al menos 2 acciones de compresión (crashing)
		\item Si cumple, ¿cuántos días de holgura tiene el proyecto?
	\end{enumerate}
